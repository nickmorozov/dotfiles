    80→                this.isTemplateDisabled = false;
    81→            }
    82→
    83→            // Map resources to picklist options

    85→                label: option.label,
    86→                value: option.value
    87→            }));
    88→
    89→            // Auto-select and disable if only one resource
    90→            if (this.resourceOptions.length === 1) {
    91→                this.ownerId = this.resourceOptions[0].value;
    92→                this.isOwnerDisabled = true;
    93→            } else {
    94→                this.isOwnerDisabled = false;
    95→            }
    96→        } catch (error) {
    97→            console.error('Error loading picklist data:', error);
    98→            this.showError('Error loading form data: ' + (error.body?.message || error.message));
    99→        } finally {
   100→            this.isLoadingData = false;
   101→        }
   102→    }

   166→    async handleSave() {

   168→        if (!this.workItemName || this.workItemName.trim().length === 0) {

   172→

   174→

   177→                projectId: this.projectId,
   178→                parentWorkItemId: this.parentWorkItemId,
   179→                templateId: this.templateId,
   180→                name: this.workItemName.trim(),
   181→                summary: this.summary,
   182→                nextSteps: this.nextSteps,
   183→                startDate: this.startDate,
   184→                dueDate: this.dueDate,

   186→                ownerId: this.ownerId
   187→            };
   188→
   189→            console.log('Creating work item with request:', JSON.stringify(request));
   190→            const workItemId = await createWorkItem({ request });

   192→            // Close modal first to avoid event issues

   194→            //console.log('Work item create modal closed after creation');

   211→            const errorMessage = error.body?.message || error.message || 'Unknown error creating work item';
   212→            console.error('Full error details:', JSON.stringify(error));
   213→            this.showError(errorMessage, 'Error Creating Work Item');
