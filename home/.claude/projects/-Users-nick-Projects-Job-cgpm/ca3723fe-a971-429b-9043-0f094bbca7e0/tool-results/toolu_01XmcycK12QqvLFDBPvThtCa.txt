    27→        public ProjectDataWrapper(Id projectId, Date startDate, Date dueDate, List<WorkItemWrapper> items) {
    28→            this.projectId = projectId;
    29→            this.projectStartDate = startDate;
    30→            this.projectDueDate = dueDate;

    33→            this.statusNames = new Set<String>();
    34→            this.getStatusOptions(items);
    35→
    36→            this.statusOptions = new List<PicklistOption>{ new PicklistOption('All Statuses', '') };
    37→            for (String statusName : statusNames) {
    38→                this.statusOptions.add(new PicklistOption(statusName, statusName));
    39→            }
    40→        }
    41→
    42→        private void getStatusOptions(List<WorkItemWrapper> items) {
    43→            for (WorkItemWrapper item : items) {
    44→                if (item.workflow != null) {
    45→                    for (StatusOption option : item.workflow) {
    46→                        statusNames.add(option.name);
    47→                    }
    48→                }
    49→
    50→                if (item.children != null) {
    51→                    this.getStatusOptions(item.children);
    52→                }
    53→            }
    54→        }

   279→     * Gets all root-level Work Items for a Project with their children
   280→     *
   281→     * @param projectId The Project ID to query Work Items for
   282→     * @param maxDepth Maximum hierarchy depth to retrieve (default 3)
   283→     *
   284→     * @return ProjectDataWrapper with project dates and work items
   285→     */
   286→    @AuraEnabled(Cacheable=true)
   287→    public static ProjectDataWrapper getWorkItemsForProject(Id projectId, Integer maxDepth) {
   288→        try {
   289→            if (maxDepth == null || maxDepth < 1) {
   290→                maxDepth = 3;
   291→            }
   292→
   293→            // Get project dates
   294→            Project__c project = [
   295→                SELECT Id, Start_Date__c, End_Date__c
   296→                FROM Project__c
   297→                WHERE Id = :projectId
   298→                WITH USER_MODE
   299→                LIMIT 1
   300→            ];
   301→
   302→            List<Work_Item__c> rootItems = WorkItemService.getWorkItemsForProject(projectId, maxDepth);
   303→
   304→            // Convert to wrappers
   305→            List<WorkItemWrapper> rootWrappers = new List<WorkItemWrapper>();
   306→            for (Work_Item__c item : rootItems) {
   307→                rootWrappers.add(new WorkItemWrapper(item, 0));
   308→            }
   309→
   310→            // Build hierarchy if depth > 0
   311→            if (maxDepth > 0 && !rootWrappers.isEmpty()) {
   312→                Set<Id> rootIds = new Set<Id>();
   313→                for (WorkItemWrapper wrapper : rootWrappers) {
   314→                    rootIds.add(wrapper.id);
   315→                }
   316→                buildHierarchy(rootWrappers, rootIds, 1, maxDepth);
   317→            }
   318→
   319→            return new ProjectDataWrapper(project.Id, project.Start_Date__c, project.End_Date__c, rootWrappers);
   320→        } catch (Exception e) {
   321→            System.debug(LoggingLevel.ERROR, 'Error in getWorkItemsForProject: ' + e.getMessage());
   322→            throw new AppException('Error retrieving Work Items: ' + e.getMessage());
   323→        }
   324→    }
