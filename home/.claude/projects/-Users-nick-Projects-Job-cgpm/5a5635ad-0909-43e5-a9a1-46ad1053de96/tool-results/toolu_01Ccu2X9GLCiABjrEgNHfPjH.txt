     1→import { api, track, wire } from 'lwc';
     2→import { refreshApex } from '@salesforce/apex';
     3→import getAvailableResources from '@salesforce/apex/WorkItemController.getAvailableResources';
     4→import getProjectRoles from '@salesforce/apex/WorkItemController.getProjectRoles';
     5→import refreshWorkItem from '@salesforce/apex/WorkItemController.refreshWorkItem';
     6→import addWorkItemAssignment from '@salesforce/apex/WorkItemController.addWorkItemAssignment';
     7→import removeWorkItemAssignment from '@salesforce/apex/WorkItemController.removeWorkItemAssignment';
     8→import updateWorkItemOwner from '@salesforce/apex/WorkItemController.updateWorkItemOwner';
     9→import BaseComponent from 'c/baseComponent';
    10→
    11→export default class ResourceAssignmentModal extends BaseComponent {
    12→    @api workItemId;
    13→    @api workItemName;
    14→    @api projectId;
    15→
    16→    @track _currentAssignments = [];
    17→    @track _ownerId;
    18→    @track _ownerName;
    19→    @track _isOpen = false;
    20→    @track isLoading = false;
    21→    @track isEditingOwner = false;
    22→    @track selectedResourceId;
    23→    @track selectedProjectRoleId;
    24→    @track selectedOwnerId;
    25→
    26→    availableResources = [];
    27→    projectRoles = [];
    28→    wiredResourcesResult;
    29→    wiredProjectRolesResult;
    30→
    31→    // Track if modal should stay open after data refresh
    32→    shouldStayOpen = false;
    33→
    34→    @api
    35→    get currentAssignments() {
    36→        return this._currentAssignments;
    37→    }
    38→    set currentAssignments(value) {
    39→        this._currentAssignments = value || [];
    40→    }
    41→
    42→    @api
    43→    get ownerId() {
    44→        return this._ownerId;
    45→    }
    46→    set ownerId(value) {
    47→        this._ownerId = value;
    48→    }
    49→
    50→    @api
    51→    get ownerName() {
    52→        return this._ownerName;
    53→    }
    54→    set ownerName(value) {
    55→        this._ownerName = value;
    56→    }
    57→
    58→    get isOpen() {
    59→        return this._isOpen;
    60→    }
    61→
    62→    /**
    63→     * Wire to get available resources for the project
    64→     */
    65→    @wire(getAvailableResources, { projectId: '$projectId' })
    66→    wiredResources(result) {
    67→        this.wiredResourcesResult = result;
    68→        const { data, error } = result;
    69→
    70→        if (data) {
    71→            this.availableResources = data;
    72→        } else if (error) {
    73→            this.showError('Failed to load available resources');
    74→            console.error('Error loading resources:', error);
    75→        }
    76→    }
    77→
    78→    /**
    79→     * Wire to get project roles
    80→     */
    81→    @wire(getProjectRoles, { projectId: '$projectId' })
    82→    wiredProjectRoles(result) {
    83→        this.wiredProjectRolesResult = result;
    84→        const { data, error } = result;
    85→
    86→        if (data) {
    87→            this.projectRoles = data;
    88→        } else if (error) {
    89→            this.showError('Failed to load project roles');
    90→            console.error('Error loading project roles:', error);
    91→        }
    92→    }
    93→
    94→    /**
    95→     * Open the modal
    96→     */
    97→    @api
    98→    async open() {
    99→        this._isOpen = true;
   100→        this.shouldStayOpen = false;
   101→        this.resetForm();
   102→
   103→        // Refresh work item data when opening to ensure we have latest assignments
   104→        await this.refreshWorkItemData();
   105→    }
   106→
   107→    /**
   108→     * Close the modal
   109→     */
   110→    @api
   111→    close() {
   112→        this._isOpen = false;
   113→        this.shouldStayOpen = false;
   114→        this.isEditingOwner = false;
   115→        this.resetForm();
   116→    }
   117→
   118→    /**
   119→     * Handle close button click
   120→     */
   121→    handleClose() {
   122→        // Trigger a final refresh of the parent component before closing
   123→        this.dispatchEvent(
   124→            new CustomEvent('modalclosed', {
   125→                detail: { workItemId: this.workItemId },
   126→                bubbles: true,
   127→                composed: true
   128→            })
   129→        );
   130→
   131→        this.close();
   132→    }
   133→
   134→    /**
   135→     * Called when parent component data refreshes
   136→     */
   137→    renderedCallback() {
   138→        // If modal should stay open and it got closed due to data refresh, reopen it
   139→        if (this.shouldStayOpen && !this._isOpen) {
   140→            this._isOpen = true;
   141→        }
   142→    }
   143→
   144→    /**
   145→     * Handle edit owner button click
   146→     */
   147→    handleEditOwner() {
   148→        this.isEditingOwner = true;
   149→        this.selectedOwnerId = this._ownerId;
   150→    }
   151→
   152→    /**
   153→     * Handle cancel owner edit
   154→     */
   155→    handleCancelOwnerEdit() {
   156→        this.isEditingOwner = false;
   157→        this.selectedOwnerId = null;
   158→    }
   159→
   160→    /**
   161→     * Handle owner selection change
   162→     */
   163→    handleOwnerChange(event) {
   164→        this.selectedOwnerId = event.detail.value;
   165→    }
   166→
   167→    /**
   168→     * Handle save owner
   169→     */
   170→    async handleSaveOwner() {
   171→        if (!this.selectedOwnerId) {
   172→            this.showWarning('Please select a new owner', 'Missing Information');
   173→            return;
   174→        }
   175→
   176→        this.isLoading = true;
   177→        this.shouldStayOpen = true; // Mark that modal should stay open
   178→
   179→        try {
   180→            await updateWorkItemOwner({
   181→                workItemId: this.workItemId,
   182→                newOwnerId: this.selectedOwnerId
   183→            });
   184→
   185→            this.showSuccess('Primary resource (owner) updated successfully');
   186→            this.isEditingOwner = false;
   187→            this.selectedOwnerId = null;
   188→
   189→            // Refresh the work item data to update displays
   190→            await this.refreshWorkItemData();
   191→
   192→            // Refresh available resources
   193→            await refreshApex(this.wiredResourcesResult);
   194→        } catch (error) {
   195→            this.shouldStayOpen = false; // Don't stay open on error
   196→            this.showError(error.body?.message || 'Failed to update owner');
   197→            console.error('Error updating owner:', error);
   198→        } finally {
   199→            this.isLoading = false;
   200→        }
   201→    }
   202→
   203→    /**
   204→     * Handle resource selection change
   205→     */
   206→    handleResourceChange(event) {
   207→        this.selectedResourceId = event.detail.value;
   208→
   209→        // Auto-populate project role from selected resource's default role
   210→        if (this.selectedResourceId && this.availableResources) {
   211→            const selectedResource = this.availableResources.find((r) => r.id === this.selectedResourceId);
   212→            if (selectedResource && selectedResource.defaultRoleId) {
   213→                this.selectedProjectRoleId = selectedResource.defaultRoleId;
   214→            }
   215→        }
   216→    }
   217→
   218→    /**
   219→     * Handle project role selection change
   220→     */
   221→    handleProjectRoleChange(event) {
   222→        this.selectedProjectRoleId = event.detail.value;
   223→    }
   224→
   225→    /**
   226→     * Handle add assignment
   227→     */
   228→    async handleAddAssignment() {
   229→        if (!this.selectedResourceId) {
   230→            this.showWarning('Please select a resource', 'Missing Information');
   231→            return;
   232→        }
   233→
   234→        this.isLoading = true;
   235→        this.shouldStayOpen = true; // Mark that modal should stay open
   236→
   237→        try {
   238→            await addWorkItemAssignment({
   239→                workItemId: this.workItemId,
   240→                resourceAssignmentId: this.selectedResourceId,
   241→                role: null,
   242→                isPrimary: false,
   243→                projectRoleId: this.selectedProjectRoleId
   244→            });
   245→
   246→            this.showSuccess('Resource assignment added successfully');
   247→            this.resetForm();
   248→
   249→            // Refresh the work item data to update displays
   250→            await this.refreshWorkItemData();
   251→
   252→            // Refresh available resources
   253→            await refreshApex(this.wiredResourcesResult);
   254→        } catch (error) {
   255→            this.shouldStayOpen = false; // Don't stay open on error
   256→            this.showError(error.body?.message || 'Failed to add resource assignment');
   257→            console.error('Error adding assignment:', error);
   258→        } finally {
   259→            this.isLoading = false;
   260→        }
   261→    }
   262→
   263→    /**
   264→     * Handle remove assignment
   265→     */
   266→    async handleRemoveAssignment(event) {
   267→        const assignmentId = event.currentTarget.dataset.assignmentId;
   268→
   269→        if (!assignmentId) {
   270→            return;
   271→        }
   272→
   273→        this.isLoading = true;
   274→        this.shouldStayOpen = true; // Mark that modal should stay open
   275→
   276→        try {
   277→            await removeWorkItemAssignment({ assignmentId });
   278→
   279→            this.showSuccess('Resource assignment removed successfully');
   280→
   281→            // Refresh the work item data to update displays
   282→            await this.refreshWorkItemData();
   283→
   284→            // Refresh available resources
   285→            await refreshApex(this.wiredResourcesResult);
   286→        } catch (error) {
   287→            this.shouldStayOpen = false; // Don't stay open on error
   288→            this.showError(error.body?.message || 'Failed to remove resource assignment');
   289→            console.error('Error removing assignment:', error);
   290→        } finally {
   291→            this.isLoading = false;
   292→        }
   293→    }
   294→
   295→    /**
   296→     * Refresh work item data to update modal display
   297→     */
   298→    async refreshWorkItemData() {
   299→        try {
   300→            const updatedWorkItem = await refreshWorkItem({
   301→                workItemId: this.workItemId
   302→            });
   303→
   304→            // Update the modal's internal tracked properties
   305→            this._ownerId = updatedWorkItem.ownerId;
   306→            this._ownerName = updatedWorkItem.ownerName;
   307→            this._currentAssignments = updatedWorkItem.assignments || [];
   308→
   309→            // Update the parent card's data by dispatching an event
   310→            this.dispatchEvent(
   311→                new CustomEvent('workitemdatarefreshed', {
   312→                    detail: { workItem: updatedWorkItem },
   313→                    bubbles: true,
   314→                    composed: true
   315→                })
   316→            );
   317→        } catch (error) {
   318→            console.error('Error refreshing work item data:', error);
   319→        }
   320→    }
   321→
   322→    /**
   323→     * Reset form fields
   324→     */
   325→    resetForm() {
   326→        this.selectedResourceId = null;
   327→        this.selectedProjectRoleId = null;
   328→    }
   329→
   330→    // === Computed Properties ===
   331→
   332→    /**
   333→     * Whether there are additional assignments (excluding owner)
   334→     */
   335→    get hasAdditionalAssignments() {
   336→        return this._currentAssignments && this._currentAssignments.length > 0;
   337→    }
   338→
   339→    /**
   340→     * Available resource options for combobox
   341→     */
   342→    get availableResourceOptions() {
   343→        if (!this.availableResources || this.availableResources.length === 0) {
   344→            return [];
   345→        }
   346→
   347→        return this.availableResources.map((resource) => ({
   348→            label: resource.resourceName,
   349→            value: resource.id
   350→        }));
   351→    }
   352→
   353→    /**
   354→     * Available user options for owner change (from resources that are Users)
   355→     */
   356→    get availableUserOptions() {
   357→        if (!this.availableResources || this.availableResources.length === 0) {
   358→            return [];
   359→        }
   360→
   361→        // Filter to only User-based resource assignments and map to User IDs
   362→        return this.availableResources
   363→            .filter((resource) => resource.userId != null)
   364→            .map((resource) => ({
   365→                label: resource.resourceName,
   366→                value: resource.userId
   367→            }));
   368→    }
   369→
   370→    /**
   371→     * Project role options for combobox
   372→     */
   373→    get projectRoleOptions() {
   374→        if (!this.projectRoles || this.projectRoles.length === 0) {
   375→            return [];
   376→        }
   377→
   378→        return this.projectRoles.map((role) => ({
   379→            label: role.Name,
   380→            value: role.Id
   381→        }));
   382→    }
   383→
   384→    /**
   385→     * Whether add button should be disabled
   386→     */
   387→    get isAddDisabled() {
   388→        return !this.selectedResourceId || this.isLoading;
   389→    }
   390→
   391→    /**
   392→     * Whether save owner button should be disabled
   393→     */
   394→    get isSaveOwnerDisabled() {
   395→        return !this.selectedOwnerId || this.selectedOwnerId === this._ownerId || this.isLoading;
   396→    }
   397→}
