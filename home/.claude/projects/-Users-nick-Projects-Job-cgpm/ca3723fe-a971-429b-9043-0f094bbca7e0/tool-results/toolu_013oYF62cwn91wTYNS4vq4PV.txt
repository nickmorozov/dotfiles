     1→import { LightningElement, api, track } from 'lwc';
     2→import { ShowToastEvent } from 'lightning/platformShowToastEvent';
     3→import { reduceErrors } from 'c/ldsUtils';
     4→import getPromotionInfo from '@salesforce/apex/CG_PromotionInterfaceController.getPromotionInfo';
     5→import statusField from '@salesforce/schema/cgcloud__Promotion__c.CG_Status__c';
     6→import phaseField from '@salesforce/schema/cgcloud__Promotion__c.cgcloud__Phase__c';
     7→
     8→// cgcloud status values
     9→const LOADING = 'LOADING';
    10→const SAVING = 'SAVING';
    11→const VALID = 'VALID';
    12→const DISABLED = 'DISABLED';
    13→
    14→// Custom status values
    15→const PREPARATION = 'Preparation';
    16→const PLANNING = 'Planning';
    17→const INACTIVE_STATUSES = ['Rejected', 'Cancelled'];
    18→
    19→export default class CGPromotionInterface extends LightningElement {
    20→    /**
    21→     * @type {string}
    22→     */
    23→    @api recordId;
    24→
    25→    /**
    26→     * @type {object}
    27→     */
    28→    @track promo = {};
    29→
    30→    /**
    31→     * @type {object[]}
    32→     */
    33→    @track fields = [];
    34→
    35→    showSpinner = true;
    36→    readOnly = true;
    37→
    38→    get status() {
    39→        return this.promo?.[statusField.fieldApiName];
    40→    }
    41→
    42→    set status(value) {
    43→        this.promo[statusField.fieldApiName] = value;
    44→    }
    45→
    46→    get phase() {
    47→        return this.promo?.[phaseField.fieldApiName];
    48→    }
    49→
    50→    set phase(value) {
    51→        this.promo[phaseField.fieldApiName] = value;
    52→    }
    53→
    54→    get restricted() {
    55→        return this.status && this.status !== PLANNING;
    56→    }
    57→
    58→    get isActiveDisabled() {
    59→        return this.readOnly || !INACTIVE_STATUSES.includes(this.status);
    60→    }
    61→
    62→    /* Data Load */
    63→
    64→    async init() {
    65→        try {
    66→            await this.loadPromo();
    67→        } catch (error) {
    68→            console.error('Error in initialization:', reduceErrors(error).join(', \r\n'), JSON.stringify(error));
    69→            this.showToast('Error', reduceErrors(error).join(', \r\n') || 'Failed to initialize promotion component', 'error');
    70→        }
    71→
    72→        this.showSpinner = false;
    73→    }
    74→
    75→    /**
    76→     * @param {string} title
    77→     * @param {string} message
    78→     * @param {string} variant
    79→     */
    80→    showToast(title, message, variant = 'info') {
    81→        this.dispatchEvent(new ShowToastEvent({ title, message, variant }));
    82→    }
    83→
    84→    async loadPromo() {
    85→        if (!this.recordId && !this.promo.Id) {
    86→            return;
    87→        }
    88→
    89→        const promotionInfo = await getPromotionInfo({ recordId: this.recordId ?? this.promo.Id });
    90→
    91→        if (!promotionInfo) {
    92→            throw new Error('No promotion info returned from server');
    93→        }
    94→
    95→        this.fields = this.disableFields(promotionInfo.fieldConfiguration);
    96→
    97→        const service = this.refs.tpm;
    98→
    99→        for (const field in promotionInfo.record) {
   100→            if (!Object.prototype.hasOwnProperty.call(promotionInfo.record, field) || this.promo[field] !== undefined) {
   101→                continue;
   102→            }
   103→
   104→            service.setPromotionField(field, promotionInfo.record[field]);
   105→        }
   106→
   107→        service.setPromotionField('$UserTemplate', promotionInfo.userTemplate);
   108→        service.setPromotionField('Owner', {
   109→            Id: promotionInfo.record.OwnerId,
   110→            $UserTemplate: promotionInfo.ownerTemplate
   111→        });
   112→        service.setPromotionField('tacticTemplates', this.disableTemplateFields(promotionInfo.tacticTemplates));
   113→        service.setPromotionField('sidebarTemplates', this.disableTemplateFields(promotionInfo.sidebarTemplates));
   114→    }
   115→
   116→    disableTemplateFields(templates) {
   117→        for (const template of [...templates]) {
   118→            template.fieldConfiguration = this.disableFields(template.fieldConfiguration);
   119→        }
   120→
   121→        return templates;
   122→    }
   123→
   124→    disableFields(fields) {
   125→        for (const field of [...fields]) {
   126→            if (this.restricted && field.fieldName !== statusField.fieldApiName) {
   127→                field.disabled = true;
   128→            }
   129→        }
   130→
   131→        return fields;
   132→    }
   133→
   134→    /* Handlers */
   135→
   136→    onPromotionChange(e) {
   137→        this.promo = { ...e.detail.value };
   138→    }
   139→
   140→    onEditModeChange(e) {
   141→        this.readOnly = !e.detail.value;
   142→
   143→        if (!this.readOnly) {
   144→            if (this.phase === PREPARATION) {
   145→                this.phase = PLANNING;
   146→
   147→                if (!this.status) {
   148→                    this.status = PLANNING;
   149→                }
   150→            }
   151→
   152→            if (this.restricted) {
   153→                this.showToast('Warning', `Promotion is not in ${PLANNING} status. Changes will be restricted.`, 'warning');
   154→            }
   155→        }
   156→    }
   157→
   158→    async onStatusChange(e) {
   159→        const oldValue = e.detail.oldValue;
   160→        const value = e.detail.value;
   161→
   162→        // Promotion finished loading
   163→        if (oldValue === LOADING && value === VALID) {
   164→            await this.init();
   165→        }
   166→
   167→        // Promotion finished saving
   168→        if (oldValue === SAVING && value === LOADING) {
   169→            this.fields = []; // clear fields to force reload
   170→            this.refs.tpm.setEditMode(false);
   171→        }
   172→    }
   173→
   174→    onCalculationStatusChange(e) {
   175→        if (e.detail.oldValue === LOADING && (e.detail.value === VALID || e.detail.value === DISABLED)) {
   176→            this.showSpinner = false;
   177→        }
   178→    }
   179→
   180→    onChange(e) {
   181→        let field = e.target.dataset.field;
   182→        let value = e.target.value;
   183→
   184→        if (e.target.type === 'checkbox' || e.target.type === 'toggle') {
   185→            value = e.target.checked;
   186→        } else if (e.target.type === 'number') {
   187→            value = Number(e.target.value);
   188→        }
   189→
   190→        this.refs.tpm.setPromotionField(field, value);
   191→    }
   192→
   193→    onFieldChange(e) {
   194→        const { fieldName, value } = e.detail;
   195→
   196→        // Handle date clamping logic
   197→        this.handleDateClamping(fieldName, value);
   198→
   199→        this.refs.tpm.setPromotionField(fieldName, value);
   200→    }
   201→
   202→    /**
   203→     * @param {string | null} field
   204→     * @param {any} value
   205→     */
   206→    /**
   207→     * @param {string | null} field
   208→     * @param {any} value
   209→     */
   210→    handleDateClamping(field, value) {
   211→        const isFrom = field === 'cgcloud__Date_From__c';
   212→        if (!isFrom && field !== 'cgcloud__Date_Thru__c') {
   213→            return;
   214→        }
   215→
   216→        const promoClampMap = {
   217→            cgcloud__Delivery_Date_From__c: 'cgcloud__Delivery_Date_Thru__c',
   218→            cgcloud__Placement_Date_From__c: 'cgcloud__Placement_Date_Thru__c',
   219→            cgcloud__Order_Date_From__c: 'cgcloud__Order_Date_Thru__c',
   220→            cgcloud__Commit_Date__c: null,
   221→            cgcloud__Reference_Date__c: null
   222→        };
   223→
   224→        const tacticClampMap = {
   225→            cgcloud__Date_From__c: 'cgcloud__Date_Thru__c',
   226→            cgcloud__Shipment_Date_From__c: 'cgcloud__Shipment_Date_Thru__c',
   227→            cgcloud__Instore_Date_From__c: 'cgcloud__Instore_Date_Thru__c'
   228→        };
   229→
   230→        // Reusable clamp engine
   231→        const runClamp = (record, map, setFn) => {
   232→            Object.entries(map).forEach(([f1, f2]) => {
   233→                [f1, f2].forEach((f) => {
   234→                    // If field exists AND (isFrom + value is greater) OR (!isFrom + value is smaller)
   235→                    if (f && (isFrom ? record[f] < value : record[f] > value)) {
   236→                        setFn(f, value);
   237→                    }
   238→                });
   239→            });
   240→        };
   241→
   242→        // 1. Process Promo
   243→        runClamp(this.promo, promoClampMap, (f, v) => this.refs.tpm.setPromotionField(f, v));
   244→
   245→        // 2. Process Tactics
   246→        const tactics = this.refs.tpm.getProperty('tactics') || [];
   247→        tactics.forEach((tactic) => runClamp(tactic, tacticClampMap, (f, v) => this.refs.tpm.setTacticField(tactic.Id, f, v)));
   248→    }
   249→}
