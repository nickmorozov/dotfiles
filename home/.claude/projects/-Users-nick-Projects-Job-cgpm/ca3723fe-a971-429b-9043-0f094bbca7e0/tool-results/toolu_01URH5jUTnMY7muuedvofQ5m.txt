     5→import { api, wire } from 'lwc';
     6→import BaseComponent from 'c/baseComponent';
     7→
     8→import getWorkItemsForProject from '@salesforce/apex/WorkItemController.getWorkItemsForProject';
     9→import getWorkItemWithChildren from '@salesforce/apex/WorkItemController.getWorkItemWithChildren';
    10→import getWorkItemTemplates from '@salesforce/apex/WorkItemCreateController.getWorkItemTemplates';
    11→import moveWorkItem from '@salesforce/apex/WorkItemController.moveWorkItem';
    12→import { getFieldValue, getRecord } from 'lightning/uiRecordApi';
    13→
    14→import WORK_ITEM_PROJECT_FIELD from '@salesforce/schema/Work_Item__c.Project__c';
    15→import { refreshApex } from '@salesforce/apex';
    16→
    17→/* * ======================================================================================
    18→ * SHARED STATE MANAGEMENT (Module Level)
    19→ * This allows all instances of DataManager to share the same data source of truth.
    20→ * ======================================================================================
    21→ */
    22→const PROJECT_DATA_CACHE = {}; // Key: projectId, Value: { workItems: [], statusOptions: [] }
    23→const INSTANCES = new Set(); // Track all active DataManager instances
    24→
    25→/**
    26→ * Clears the module-level cache for test isolation.
    27→ * Call this in beforeEach() to ensure tests don't share state.
    28→ * @export
    29→ */
    30→export function clearCache() {
    31→    Object.keys(PROJECT_DATA_CACHE).forEach((key) => delete PROJECT_DATA_CACHE[key]);
    32→    INSTANCES.clear();
    33→}
    34→
    35→/**
    36→ * Broadcast data to all DataManager instances for a specific project
    37→ */
    38→const broadcastDataChange = (projectId, data) => {
    39→    // 1. Update Cache
    40→    PROJECT_DATA_CACHE[projectId] = data;
    41→
    42→    // 2. Notify all subscribers for this project
    43→    for (const instance of INSTANCES) {
    44→        if (instance.effectiveProjectId === projectId) {
    45→            instance.notifyDataLoaded(data);
    46→        }
    47→    }
    48→};
    49→
    50→export default class DataManager extends BaseComponent {
    51→    @api recordId; // Project ID or Work Item ID from record page
    52→    @api objectApiName; // Object type (cgpm__Project__c or cgpm__Work_Item__c)
    53→    @api hierarchyDepth = 3; // Maximum hierarchy depth to load
    54→
    55→    projectId; // Resolved Project ID (either from recordId or from Work Item's project lookup)
    56→
    57→    connectedCallback() {
    58→        // Register this instance to receive broadcasts
    59→        INSTANCES.add(this);
    60→
    61→        // Immediate Check: Do we already have data in memory?
    62→        // If yes, load it immediately to prevent "pop-in" or double loading
    63→        if (this.effectiveProjectId && PROJECT_DATA_CACHE[this.effectiveProjectId]) {
    64→            this.notifyDataLoaded(PROJECT_DATA_CACHE[this.effectiveProjectId]);
    65→        }
    66→
    67→        // todo: run wires imperatively, add loading/saving state
    68→    }
    69→
    70→    disconnectedCallback() {
    71→        INSTANCES.delete(this);
    72→    }
    73→
    74→    get isProjectPage() {
    75→        return this.objectApiName === 'cgpm__Project__c';
    76→    }
    77→
    78→    get isWorkItemPage() {
    79→        return this.objectApiName === 'cgpm__Work_Item__c';
    80→    }
    81→
    82→    get effectiveProjectId() {
    83→        // On Project page, use recordId directly
    84→        // On Work Item page, use resolved projectId from lookup
    85→        return this.isProjectPage ? this.recordId : (this.projectId ?? undefined);
    86→    }
    87→
    88→    get parentWorkItemIdForTemplates() {
    89→        // On Project page: null (top-level templates)
    90→        // On Work Item page: current work item ID (child templates)
    91→        return this.isWorkItemPage ? this.recordId : undefined;
    92→    }
    93→
    94→    // Wire to get Work Item record (only when on Work Item page)
    95→    @wire(getRecord, {
    96→        recordId: '$recordId',
    97→        fields: [WORK_ITEM_PROJECT_FIELD]
    98→    })
    99→    wiredWorkItemRecord({ data, error }) {
   100→        if (this.isWorkItemPage) {
   101→            if (data) {
   102→                this.projectId = getFieldValue(data, WORK_ITEM_PROJECT_FIELD);
   103→            } else if (error) {
   104→                this.errorMessage = this.extractErrorMessage(error);
   105→                this.showError(this.errorMessage, 'Error Loading Work Item');
   106→            }
   107→        }
   108→    }
   109→
   110→    // Wire to get available work item templates
   111→    @wire(getWorkItemTemplates, {
   112→        projectId: '$effectiveProjectId',
   113→        parentWorkItemId: '$parentWorkItemIdForTemplates'
   114→    })
   115→    wiredTemplates({ data, error }) {
   116→        let templates = [];
   117→
   118→        if (data) {
   119→            templates = data;
   120→        } else if (error) {
   121→            console.error('Error loading work item templates:', error);
   122→        }
   123→
   124→        this.dispatchEvent(new CustomEvent('templatesload', { detail: { templates } }));
   125→    }
   126→
   127→    wiredWorkItemsResult = {};
   128→
   129→    // Wire method to load Work Items for the Project (when on Project page)
   130→    @wire(getWorkItemsForProject, {
   131→        projectId: '$effectiveProjectId',
   132→        maxDepth: '$hierarchyDepth'
   133→    })
   134→    wiredWorkItemsForProject(result) {
   135→        if (this.isProjectPage) {
   136→            this.wiredWorkItemsResult = result;
   137→            const { data, error } = result;
   138→
   139→            if (error) {
   140→                this.errorMessage = this.extractErrorMessage(error);
   141→                this.showError(this.errorMessage, 'Error Loading Work Items');
   142→                return;
   143→            } else if (!data) {
   144→                this.showError('No Work Items found for this Project', 'Error Loading Work Items');
   145→                return;
   146→            }
   147→
   148→            const workItems = data.workItems || [];
   149→            const statusOptions = data.statusOptions || this.defaultStatusOptions;
   150→            const newData = { workItems, statusOptions };
   151→
   152→            // DEDUPLICATION LOGIC
   153→            const projectId = this.effectiveProjectId;
   154→            const cached = PROJECT_DATA_CACHE[projectId];
   155→
   156→            // Simple string comparison to avoid broadcast loops
   157→            // If data is identical to what we already have, DO NOT broadcast.
   158→            if (cached && JSON.stringify(cached) === JSON.stringify(newData)) {
   159→                return;
   160→            }
   161→
   162→            // Update Source of Truth and Notify All
   163→            broadcastDataChange(projectId, newData);
   164→        }
   165→    }
   166→
   167→    // Wire method to load Work Item with Children (when on Work Item page)
   168→    @wire(getWorkItemWithChildren, {
   169→        workItemId: '$recordId',
   170→        maxDepth: '$hierarchyDepth'
   171→    })
   172→    wiredWorkItemWithChildren(result) {
   173→        if (this.isWorkItemPage) {
   174→            this.wiredWorkItemsResult = result;
   175→            const { data, error } = result;
   176→
   177→            if (error) {
   178→                this.errorMessage = this.extractErrorMessage(error);
   179→                this.showError(this.errorMessage, 'Error Loading Work Item');
   180→                return;
   181→            } else if (!data) {
   182→                this.showError('Work Item not found', 'Error Loading Work Item');
   183→                return;
   184→            }
   185→
   186→            const workItems = [data];
   187→            const statusOptions = this.defaultStatusOptions;
   188→
   189→            const newData = { workItems, statusOptions };
   190→
   191→            // DEDUPLICATION LOGIC
   192→            const contextId = this.recordId;
   193→            const cached = PROJECT_DATA_CACHE[contextId];
   194→
   195→            // Simple string comparison to avoid broadcast loops
   196→            // If data is identical to what we already have, DO NOT broadcast.
   197→            if (cached && JSON.stringify(cached) === JSON.stringify(newData)) {
   198→                return;
   199→            }
   200→
   201→            // Update Source of Truth and Notify All
   202→            broadcastDataChange(contextId, newData);
   203→        }
   204→    }
   205→
   206→    /**
   207→     * Internal method to dispatch the event to the parent
   208→     * Called by the broadcast system or local wire
   209→     */
   210→    @api
   211→    notifyDataLoaded(data) {
   212→        // Defensive copy to prevent mutation bugs
   213→        const detail = JSON.parse(JSON.stringify(data));
   214→        this.dispatchEvent(new CustomEvent('workitemsload', { detail }));
   215→    }
   216→
   217→    @api async refresh() {
   218→        try {
   219→            await refreshApex(this.wiredWorkItemsResult);
   220→        } catch (error) {
   221→            this.errorMessage = this.extractErrorMessage(error);
   222→            this.showError(this.errorMessage, 'Error Refreshing');
   223→        }
   224→    }
   225→
   226→    @api async moveWorkItem(draggedItemId, targetItemId, draggedParentId, targetParentId) {
   227→        try {
   228→            // Call Apex to reorder
   229→            await moveWorkItem({
   230→                draggedItemId,
   231→                targetItemId,
   232→                draggedParentId,
   233→                targetParentId
   234→            });
   235→
   236→            this.showSuccess('Work Items reordered successfully');
   237→        } catch (error) {
   238→            this.errorMessage = this.extractErrorMessage(error);
   239→            this.showError(this.errorMessage, 'Error Reordering Work Items');
   240→        }
   241→
   242→        await this.refresh();
   243→    }
   244→}
