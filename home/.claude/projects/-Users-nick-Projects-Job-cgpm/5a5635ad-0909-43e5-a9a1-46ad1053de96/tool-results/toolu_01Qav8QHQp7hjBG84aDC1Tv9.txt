   415→        event.preventDefault();
   416→        event.stopPropagation();
   417→
   418→        try {
   419→            const fields = {};
   420→            fields[ID_FIELD.fieldApiName] = this.workItem.id;
   421→            fields[NAME_FIELD.fieldApiName] = this.editName;
   422→            fields[SUMMARY_FIELD.fieldApiName] = this.editSummary;
   423→            fields[NEXT_STEPS_FIELD.fieldApiName] = this.editNextSteps;
   424→            fields[START_DATE_FIELD.fieldApiName] = this.editStartDate || null;
   425→            fields[DUE_DATE_FIELD.fieldApiName] = this.editDueDate || null;
   426→            fields[ESTIMATED_HOURS_FIELD.fieldApiName] = this.editEstimatedHours || null;
   427→
   428→            const recordInput = { fields };
   429→            await updateRecord(recordInput);
   430→
   431→            // Exit edit mode
   432→            this.isEditMode = false;
   433→
   434→            // Show success toast
   435→            this.showSuccess('Work item updated successfully');
   436→
   437→            // Dispatch update event to parent to refresh data
   438→            const updateEvent = new CustomEvent('workitemupdate', {
   439→                detail: {
   440→                    workItemId: this.workItem.id
   441→                },
   442→                bubbles: true,
   443→                composed: true
   444→            });
   445→            this.dispatchEvent(updateEvent);
   446→        } catch (error) {
   447→            console.error('Error updating work item:', error);
   448→            this.showError(error.body?.message || 'Error updating work item');

   497→    /**
   498→     * Handle status change
   499→     */
   500→    async handleStatusChange(event) {
   501→        event.preventDefault();
   502→        event.stopPropagation();
   503→
   504→        const newStatus = event.currentTarget.dataset.status;
   505→
   506→        // Don't update if same status
   507→        if (newStatus === this.workItem.status) {
   508→            this.showStatusDropdown = false;
   509→            return;
   510→        }
   511→
   512→        try {
   513→            const fields = {};
   514→            fields[ID_FIELD.fieldApiName] = this.workItem.id;
   515→            fields[STATUS_FIELD.fieldApiName] = newStatus;
   516→
   517→            const recordInput = { fields };
   518→            await updateRecord(recordInput);
   519→
   520→            // Hide dropdown
   521→            this.showStatusDropdown = false;
   522→
   523→            // Show success toast
   524→            this.showSuccess('Status updated successfully');
   525→
   526→            // Dispatch update event to parent (with suppressToast to avoid duplicate)
   527→            const updateEvent = new CustomEvent('workitemupdate', {
   528→                detail: {
   529→                    workItemId: this.workItem.id,
   530→                    changes: { status: newStatus },
   531→                    suppressToast: true
   532→                },
   533→                bubbles: true,
   534→                composed: true
   535→            });
   536→            this.dispatchEvent(updateEvent);
   537→        } catch (error) {
   538→            console.error('Error updating status:', error);
   539→            this.showError(error.body?.message || 'Error updating status');
   540→        }
   541→    }

   572→    get primaryHighlight() {
   573→        if (!this.workItem.highlights || this.workItem.highlights.length === 0) {
   574→            return null;

   576→
   577→        // Return highest priority highlight
   578→        if (this.workItem.highlights.includes('blocked')) {
   579→            return 'blocked';
   580→        }
   581→        if (this.workItem.highlights.includes('at-risk')) {
   582→            return 'at-risk';
   583→        }
   584→        if (this.workItem.highlights.includes('timeline-conflict')) {
   585→            return 'timeline-conflict';
   586→        }
   587→
   588→        // Default to first highlight if no priority match
   589→        return this.workItem.highlights[0];

   605→    get dueDateClass() {
   606→        const baseClass = 'slds-text-body_regular';
   607→        const today = new Date();
   608→        today.setHours(0, 0, 0, 0);
   609→
   610→        // Parse date string as local date to avoid timezone shifts
   611→        const [year, month, day] = this.workItem.dueDate.split('-');
   612→        const dueDate = new Date(year, month - 1, day);
   613→
   614→        if (dueDate < today && this.workItem.status !== 'Complete') {
   615→            return `${baseClass} slds-text-color_error`;
   616→        }
   617→
   618→        return baseClass;
   619→    }

   730→    get timelineProgress() {

   733→        }
   734→
   735→        const today = new Date();
   736→        today.setHours(0, 0, 0, 0); // Reset to midnight for comparison
   737→
   738→        // Parse dates as local dates to avoid timezone shifts
   739→        const [startYear, startMonth, startDay] = this.workItem.startDate.split('-');
   740→        const start = new Date(startYear, startMonth - 1, startDay);
   741→
   742→        const [dueYear, dueMonth, dueDay] = this.workItem.dueDate.split('-');
   743→        const due = new Date(dueYear, dueMonth - 1, dueDay);
   744→
   745→        // If hasn't started yet, 0%
   746→        if (today < start) {
   747→            return 0;
   748→        }
   749→
   750→        // If past due date, 100%
   751→        if (today > due) {
   752→            return 100;
   753→        }
   754→
   755→        // Calculate percentage of time elapsed
   756→        const totalDuration = due.getTime() - start.getTime();
   757→        const elapsed = today.getTime() - start.getTime();
   758→        const progress = Math.round((elapsed / totalDuration) * 100);
   759→
   760→        return Math.min(Math.max(progress, 0), 100); // Clamp between 0-100
   761→    }
